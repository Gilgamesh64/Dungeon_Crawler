358/*
359 * Open directory stream DIRNAME for read and return a pointer to the
360 * internal working area that is used to retrieve individual directory
361 * entries.
362 */
363static _WDIR *
364_wopendir(const wchar_t *dirname)
365{
366 wchar_t *p;
367
368 /* Must have directory name */
369 if (dirname == NULL || dirname[0] == '\0') {
370 dirent_set_errno(ENOENT);
371 return NULL;
372 }
373
374 /* Allocate new _WDIR structure */
375 _WDIR *dirp = (_WDIR*) malloc(sizeof(struct _WDIR));
376 if (!dirp)
377 return NULL;
378
379 /* Reset _WDIR structure */
380 dirp->handle = INVALID_HANDLE_VALUE;
381 dirp->patt = NULL;
382 dirp->cached = 0;
383 dirp->invalid = 0;
384
385 /*
386 * Compute the length of full path plus zero terminator
387 *
388 * Note that on WinRT there's no way to convert relative paths
389 * into absolute paths, so just assume it is an absolute path.
390 */
391#if !defined(WINAPI_FAMILY_PARTITION) || WINAPI_FAMILY_PARTITION(WINAPI_PARTITION_DESKTOP)
392 /* Desktop */
393 DWORD n = GetFullPathNameW(dirname, 0, NULL, NULL);
394#else
395 /* WinRT */
396 size_t n = wcslen(dirname);
397#endif
398
399 /* Allocate room for absolute directory name and search pattern */
400 dirp->patt = (wchar_t*) malloc(sizeof(wchar_t) * n + 16);
401 if (dirp->patt == NULL)
402 goto exit_closedir;
403
404 /*
405 * Convert relative directory name to an absolute one. This
406 * allows rewinddir() to function correctly even when current
407 * working directory is changed between opendir() and rewinddir().
408 *
409 * Note that on WinRT there's no way to convert relative paths
410 * into absolute paths, so just assume it is an absolute path.
411 */
412#if !defined(WINAPI_FAMILY_PARTITION) || WINAPI_FAMILY_PARTITION(WINAPI_PARTITION_DESKTOP)
413 /* Desktop */
414 n = GetFullPathNameW(dirname, n, dirp->patt, NULL);
415 if (n <= 0)
416 goto exit_closedir;
417#else
418 /* WinRT */
419 wcsncpy_s(dirp->patt, n+1, dirname, n);
420#endif
421
422 /* Append search pattern \* to the directory name */
423 p = dirp->patt + n;
424 switch (p[-1]) {
425 case '\\':
426 case '/':
427 case ':':
428 /* Directory ends in path separator, e.g. c:\temp\ */
429 /*NOP*/;
430 break;
431
432 default:
433 /* Directory name doesn't end in path separator */
434 *p++ = '\\';
435 }
436 *p++ = '*';
437 *p = '\0';
438
439 /* Open directory stream and retrieve the first entry */
440 if (!dirent_first(dirp))
441 goto exit_closedir;
442
443 /* Success */
444 return dirp;
445
446 /* Failure */
447exit_closedir:
448 _wclosedir(dirp);
449 return NULL;
450}

472/*
473 * Read next directory entry.
474 *
475 * Returns zero on success. If end of directory stream is reached, then sets
476 * result to NULL and returns zero.
477 */
478static int
479_wreaddir_r(
480 _WDIR *dirp, struct _wdirent *entry, struct _wdirent **result)
481{
482 /* Validate directory handle */
483 if (!dirp || dirp->handle == INVALID_HANDLE_VALUE || !dirp->patt) {
484 dirent_set_errno(EBADF);
485 *result = NULL;
486 return -1;
487 }
488
489 /* Read next directory entry */
490 WIN32_FIND_DATAW *datap = dirent_next(dirp);
491 if (!datap) {
492 /* Return NULL to indicate end of directory */
493 *result = NULL;
494 return /*OK*/0;
495 }
496
497 /*
498 * Copy file name as wide-character string. If the file name is too
499 * long to fit in to the destination buffer, then truncate file name
500 * to PATH_MAX characters and zero-terminate the buffer.
501 */
502 size_t i = 0;
503 while (i < PATH_MAX && datap->cFileName[i] != 0) {
504 entry->d_name[i] = datap->cFileName[i];
505 i++;
506 }
507 entry->d_name[i] = 0;
508
509 /* Length of file name excluding zero terminator */
510 entry->d_namlen = i;
511
512 /* Determine file type */
513 DWORD attr = datap->dwFileAttributes;
514 if ((attr & FILE_ATTRIBUTE_DEVICE) != 0)
515 entry->d_type = DT_CHR;
516 else if ((attr & FILE_ATTRIBUTE_REPARSE_POINT) != 0)
517 entry->d_type = DT_LNK;
518 else if ((attr & FILE_ATTRIBUTE_DIRECTORY) != 0)
519 entry->d_type = DT_DIR;
520 else
521 entry->d_type = DT_REG;
522
523 /* Read the next directory entry to cache */
524 datap = dirent_next(dirp);
525 if (datap) {
526 /* Compute 31-bit hash of the next directory entry */
527 entry->d_off = dirent_hash(datap);
528
529 /* Push the next directory entry back to cache */
530 dirp->cached = 1;
531 } else {
532 /* End of directory stream */
533 entry->d_off = (long) ((~0UL) >> 1);
534 }
535
536 /* Reset other fields */
537 entry->d_ino = 0;
538 entry->d_reclen = sizeof(struct _wdirent);
539
540 /* Set result address */
541 *result = entry;
542 return /*OK*/0;
543}

545/*
546 * Close directory stream opened by opendir() function. This invalidates the
547 * DIR structure as well as any directory entry read previously by
548 * _wreaddir().
549 */
550static int
551_wclosedir(_WDIR *dirp)
552{
553 if (!dirp) {
554 dirent_set_errno(EBADF);
555 return /*failure*/-1;
556 }
557
558 /*
559 * Release search handle if we have one. Being able to handle
560 * partially initialized _WDIR structure allows us to use this
561 * function to handle errors occurring within _wopendir.
562 */
563 if (dirp->handle != INVALID_HANDLE_VALUE) {
564 FindClose(dirp->handle);
565 }
566
567 /*
568 * Release search pattern. Note that we don't need to care if
569 * dirp->patt is NULL or not: function free is guaranteed to act
570 * appropriately.
571 */
572 free(dirp->patt);
573
574 /* Release directory structure */
575 free(dirp);
576 return /*success*/0;
577}

579/*
580 * Rewind directory stream such that _wreaddir() returns the very first
581 * file name again.
582 */
583static void _wrewinddir(_WDIR* dirp)
584{
585 /* Check directory pointer */
586 if (!dirp || dirp->handle == INVALID_HANDLE_VALUE || !dirp->patt)
587 return;
588
589 /* Release existing search handle */
590 FindClose(dirp->handle);
591
592 /* Open new search handle */
593 dirent_first(dirp);
594}

596/* Get first directory entry */
597static WIN32_FIND_DATAW *
598dirent_first(_WDIR *dirp)
599{
600 /* Open directory and retrieve the first entry */
601 dirp->handle = FindFirstFileExW(
602 dirp->patt, FindExInfoStandard, &dirp->data,
603 FindExSearchNameMatch, NULL, 0);
604 if (dirp->handle == INVALID_HANDLE_VALUE)
605 goto error;
606
607 /* A directory entry is now waiting in memory */
608 dirp->cached = 1;
609 return &dirp->data;
610
611error:
612 /* Failed to open directory: no directory entry in memory */
613 dirp->cached = 0;
614 dirp->invalid = 1;
615
616 /* Set error code */
617 DWORD errorcode = GetLastError();
618 switch (errorcode) {
619 case ERROR_ACCESS_DENIED:
620 /* No read access to directory */
621 dirent_set_errno(EACCES);
622 break;
623
624 case ERROR_DIRECTORY:
625 /* Directory name is invalid */
626 dirent_set_errno(ENOTDIR);
627 break;
628
629 case ERROR_PATH_NOT_FOUND:
630 default:
631 /* Cannot find the file */
632 dirent_set_errno(ENOENT);
633 }
634 return NULL;
635}

637/* Get next directory entry */
638static WIN32_FIND_DATAW *
639dirent_next(_WDIR *dirp)
640{
641 /* Return NULL if seek position was invalid */
642 if (dirp->invalid)
643 return NULL;
644
645 /* Is the next directory entry already in cache? */
646 if (dirp->cached) {
647 /* Yes, a valid directory entry found in memory */
648 dirp->cached = 0;
649 return &dirp->data;
650 }
651
652 /* Read the next directory entry from stream */
653 if (FindNextFileW(dirp->handle, &dirp->data) == FALSE) {
654 /* End of directory stream */
655 return NULL;
656 }
657
658 /* Success */
659 return &dirp->data;
660}

662/*
663 * Compute 31-bit hash of file name.
664 *
665 * See djb2 at http://www.cse.yorku.ca/~oz/hash.html
666 */
667static long
668dirent_hash(WIN32_FIND_DATAW *datap)
669{
670 unsigned long hash = 5381;
671 unsigned long c;
672 const wchar_t *p = datap->cFileName;
673 const wchar_t *e = p + MAX_PATH;
674 while (p != e && (c = *p++) != 0) {
675 hash = (hash << 5) + hash + c;
676 }
677
678 return (long) (hash & ((~0UL) >> 1));
679}

681/* Open directory stream using plain old C-string */
682static DIR *opendir(const char *dirname)
683{
684 /* Must have directory name */
685 if (dirname == NULL || dirname[0] == '\0') {
686 dirent_set_errno(ENOENT);
687 return NULL;
688 }
689
690 /* Allocate memory for DIR structure */
691 struct DIR *dirp = (DIR*) malloc(sizeof(struct DIR));
692 if (!dirp)
693 return NULL;
694
695 /* Convert directory name to wide-character string */
696 wchar_t wname[PATH_MAX + 1];
697 size_t n;
698 int error = mbstowcs_s(&n, wname, PATH_MAX + 1, dirname, PATH_MAX+1);
699 if (error)
700 goto exit_failure;
701
702 /* Open directory stream using wide-character name */
703 dirp->wdirp = _wopendir(wname);
704 if (!dirp->wdirp)
705 goto exit_failure;
706
707 /* Success */
708 return dirp;
709
710 /* Failure */
711exit_failure:
712 free(dirp);
713 return NULL;
714}

731/*
732 * Read next directory entry into called-allocated buffer.
733 *
734 * Returns zero on success. If the end of directory stream is reached, then
735 * sets result to NULL and returns zero.
736 */
737static int
738readdir_r(
739 DIR *dirp, struct dirent *entry, struct dirent **result)
740{
741 /* Read next directory entry */
742 WIN32_FIND_DATAW *datap = dirent_next(dirp->wdirp);
743 if (!datap) {
744 /* No more directory entries */
745 *result = NULL;
746 return /*OK*/0;
747 }
748
749 /* Attempt to convert file name to multi-byte string */
750 size_t n;
751 int error = wcstombs_s(
752 &n, entry->d_name, PATH_MAX + 1,
753 datap->cFileName, PATH_MAX + 1);
754
755 /*
756 * If the file name cannot be represented by a multi-byte string, then
757 * attempt to use old 8+3 file name. This allows the program to
758 * access files although file names may seem unfamiliar to the user.
759 *
760 * Be ware that the code below cannot come up with a short file name
761 * unless the file system provides one. At least VirtualBox shared
762 * folders fail to do this.
763 */
764 if (error && datap->cAlternateFileName[0] != '\0') {
765 error = wcstombs_s(
766 &n, entry->d_name, PATH_MAX + 1,
767 datap->cAlternateFileName, PATH_MAX + 1);
768 }
769
770 if (!error) {
771 /* Length of file name excluding zero terminator */
772 entry->d_namlen = n - 1;
773
774 /* Determine file type */
775 DWORD attr = datap->dwFileAttributes;
776 if ((attr & FILE_ATTRIBUTE_DEVICE) != 0)
777 entry->d_type = DT_CHR;
778 else if ((attr & FILE_ATTRIBUTE_REPARSE_POINT) != 0)
779 entry->d_type = DT_LNK;
780 else if ((attr & FILE_ATTRIBUTE_DIRECTORY) != 0)
781 entry->d_type = DT_DIR;
782 else
783 entry->d_type = DT_REG;
784
785 /* Get offset of next file */
786 datap = dirent_next(dirp->wdirp);
787 if (datap) {
788 /* Compute 31-bit hash of the next directory entry */
789 entry->d_off = dirent_hash(datap);

792 dirp->wdirp->cached = 1;
793 } else {

795 entry->d_off = (long) ((~0UL) >> 1);
796 }

799 entry->d_ino = 0;
800 entry->d_reclen = sizeof(struct dirent);

804 * an erroneous directory entry and return that. Note that
805 * we cannot return NULL as that would stop the processing
806 * of directory entries completely.
807 */

809 entry->d_name[1] = '\0';
810 entry->d_namlen = 1;
811 entry->d_type = DT_UNKNOWN;
812 entry->d_ino = 0;
813 entry->d_off = -1;
814 entry->d_reclen = 0;
815 }

818 *result = entry;

891/* Seek directory stream to offset */
892static void
893_wseekdir(_WDIR *dirp, long loc)
894{
895 if (!dirp)
896 return;
897
898 /* Directory must be open */
899 if (dirp->handle == INVALID_HANDLE_VALUE)
900 goto exit_failure;
901
902 /* Ensure that seek position is valid */
903 if (loc < 0)
904 goto exit_failure;
905
906 /* Restart directory stream from the beginning */
907 FindClose(dirp->handle);
908 if (!dirent_first(dirp))
909 goto exit_failure;
910
911 /* Reset invalid flag so that we can read from the stream again */
912 dirp->invalid = 0;
913
914 /*
915 * Read directory entries from the beginning until the hash matches a
916 * file name. Be ware that hash code is only 31 bits longs and
917 * duplicates are possible: the hash code cannot return the position
918 * with 100.00% accuracy! Moreover, the method is slow for large
919 * directories.
920 */
921 long hash;
922 do {
923 /* Read next directory entry */
924 WIN32_FIND_DATAW *datap = dirent_next(dirp);
925 if (!datap) {
926 /*
927 * End of directory stream was reached before finding
928 * the requested location. Perhaps the file in
929 * question was deleted or moved out of the directory.
930 */
931 goto exit_failure;
932 }
933
934 /* Does the file name match the hash? */
935 hash = dirent_hash(datap);
936 } while (hash != loc);
937
938 /*
939 * File name matches the hash! Push the directory entry back to cache
940 * from where next readdir() will return it.
941 */
942 dirp->cached = 1;
943 dirp->invalid = 0;
944 return;
945
946exit_failure:
947 /* Ensure that readdir will return NULL */
948 dirp->invalid = 1;
949}

961/* Scan directory for entries */
962static int
963scandir(
964 const char *dirname, struct dirent ***namelist,
965 int (*filter)(const struct dirent*),
966 int (*compare)(const struct dirent**, const struct dirent**))
967{
968 int result;
969
970 /* Open directory stream */
971 DIR *dir = opendir(dirname);
972 if (!dir) {
973 /* Cannot open directory */
974 return /*Error*/ -1;
975 }
976
977 /* Read directory entries to memory */
978 struct dirent *tmp = NULL;
979 struct dirent **files = NULL;
980 size_t size = 0;
981 size_t allocated = 0;
982 while (1) {
983 /* Allocate room for a temporary directory entry */
984 if (!tmp) {
985 tmp = (struct dirent*) malloc(sizeof(struct dirent));
986 if (!tmp)
987 goto exit_failure;
988 }
989
990 /* Read directory entry to temporary area */
991 struct dirent *entry;
992 if (readdir_r(dir, tmp, &entry) != /*OK*/0)
993 goto exit_failure;
994
995 /* Stop if we already read the last directory entry */
996 if (entry == NULL)
997 goto exit_success;
998
999 /* Determine whether to include the entry in results */
1000 if (filter && !filter(tmp))
1001 continue;
1002
1003 /* Enlarge pointer table to make room for another pointer */
1004 if (size >= allocated) {
1005 /* Compute number of entries in the new table */
1006 size_t num_entries = size * 2 + 16;
1007
1008 /* Allocate new pointer table or enlarge existing */
1009 void *p = realloc(files, sizeof(void*) * num_entries);
1010 if (!p)
1011 goto exit_failure;
1012
1013 /* Got the memory */
1014 files = (dirent**) p;
1015 allocated = num_entries;
1016 }
1017
1018 /* Store the temporary entry to ptr table */
1019 files[size++] = tmp;
1020 tmp = NULL;
1021 }
1022
1023exit_failure:
1024 /* Release allocated entries */
1025 for (size_t i = 0; i < size; i++) {
1026 free(files[i]);
1027 }
1028
1029 /* Release the pointer table */
1030 free(files);
1031 files = NULL;
1032
1033 /* Exit with error code */
1034 result = /*error*/ -1;
1035 goto exit_status;
1036
1037exit_success:
1038 /* Sort directory entries */
1039 if (size > 1 && compare) {
1040 qsort(files, size, sizeof(void*),
1041 (int (*) (const void*, const void*)) compare);
1042 }
1043
1044 /* Pass pointer table to caller */
1045 if (namelist)
1046 *namelist = files;
1047
1048 /* Return the number of directory entries read */
1049 result = (int) size;
1050
1051exit_status:
1052 /* Release temporary directory entry, if we had one */
1053 free(tmp);
1054
1055 /* Close directory stream */
1056 closedir(dir);
1057 return result;
1058}

1074/* Compare strings */
1075static int
1076strverscmp(const char *a, const char *b)
1077{
1078 size_t i = 0;
1079 size_t j;
1080
1081 /* Find first difference */
1082 while (a[i] == b[i]) {
1083 if (a[i] == '\0') {
1084 /* No difference */
1085 return 0;
1086 }
1087 ++i;
1088 }
1089
1090 /* Count backwards and find the leftmost digit */
1091 j = i;
1092 while (j > 0 && isdigit(a[j-1])) {
1093 --j;
1094 }
1095
1096 /* Determine mode of comparison */
1097 if (a[j] == '0' || b[j] == '0') {
1098 /* Find the next non-zero digit */
1099 while (a[j] == '0' && a[j] == b[j]) {
1100 j++;
1101 }
1102
1103 /* String with more digits is smaller, e.g 002 < 01 */
1104 if (isdigit(a[j])) {
1105 if (!isdigit(b[j])) {
1106 return -1;
1107 }
1108 } else if (isdigit(b[j])) {
1109 return 1;
1110 }
1111 } else if (isdigit(a[j]) && isdigit(b[j])) {
1112 /* Numeric comparison */
1113 size_t k1 = j;
1114 size_t k2 = j;
1115
1116 /* Compute number of digits in each string */
1117 while (isdigit(a[k1])) {
1118 k1++;
1119 }
1120 while (isdigit(b[k2])) {
1121 k2++;
1122 }
1123
1124 /* Number with more digits is bigger, e.g 999 < 1000 */
1125 if (k1 < k2)
1126 return -1;
1127 else if (k1 > k2)
1128 return 1;
1129 }
1130
1131 /* Alphabetical comparison */
1132 return (int) ((unsigned char) a[i]) - ((unsigned char) b[i]);
1133}
